<!DOCTYPE html>
<html>
<head>
    <title>Personal Profile Card</title>

    <style>
        * {
            box-sizing: border-box;
        }

        body {
            margin: 0;
            height: 100vh;
            display: flex;
            justify-content: center;
            align-items: center;
            background: linear-gradient(135deg, lightblue, white);
            font-family: Arial, sans-serif;

            /* Smooth page loading */
            animation: pageFade 1.2s ease-in;
        }

        /* Profile Card */
        .card {
            width: 320px;
            padding: 30px;
            background-color: white;
            text-align: center;
            border-radius: 20px;
            box-shadow: 0 10px 25px gray;

            /* Fade in + scale animation */
            animation: cardFade 1.2s ease-out;

            /* Smooth scale transition */
            transition: transform 0.4s ease, box-shadow 0.4s ease;
        }

        /* Scale effect when mouse is over card */
        .card:hover {
            transform: scale(1.05);
            box-shadow: 0 15px 35px gray;
        }

        .card h1 {
            color: #333;
            margin-bottom: 20px;
        }

        .card p {
            color: #555;
            font-size: 16px;
            margin: 12px;
        }

        button {
            margin-top: 15px;
            padding: 12px 25px;
            border: none;
            border-radius: 8px;
            background-color: blue;
            color: white;
            font-size: 16px;
            cursor: pointer;

            transition: transform 0.3s ease, background-color 0.3s ease;
        }

        button:hover {
            background-color: darkblue;
            transform: scale(1.08);
        }

        /* Page fade animation */
        @keyframes pageFade {
            from {
                opacity: 0;
            }

            to {
                opacity: 1;
            }
        }

        /* Card fade + scale animation */
        @keyframes cardFade {
            from {
                opacity: 0;
                transform: scale(0.7);
            }

            to {
                opacity: 1;
                transform: scale(1);
            }
        }
    </style>
</head>
